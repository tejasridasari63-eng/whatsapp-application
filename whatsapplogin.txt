<script>
  const $ = id => document.getElementById(id);
  const phoneForm = $('phoneForm'), otpForm = $('otpForm'), done = $('done');
  const boxes = [...document.querySelectorAll('#otp input')];
  const DEMO_CODE = '123456'; // Replace with real server-side verification
  let countdown, secs;

  function show(el) { [phoneForm, otpForm, done].forEach(e => e.classList.toggle('hidden', e !== el)); }

  phoneForm.addEventListener('submit', e => {
    e.preventDefault();
    const num = $('phone').value.replace(/\D/g, '');
    if (num.length < 7 || num.length > 12) {
      $('phoneErr').textContent = 'Enter a valid phone number (7–12 digits).';
      return $('phone').focus();
    }
    $('phoneErr').textContent = '';
    $('sentTo').textContent = $('cc').value + ' ' + num;
    show(otpForm); boxes.forEach(b => b.value = ''); boxes[0].focus(); startTimer();
    // TODO: call your API here to send the OTP
  });

  $('phone').addEventListener('input', e => e.target.value = e.target.value.replace(/\D/g, ''));

  boxes.forEach((b, i) => {
    b.addEventListener('input', () => {
      b.value = b.value.replace(/\D/g, '');
      if (b.value && i < 5) boxes[i + 1].focus();
    });
    b.addEventListener('keydown', e => {
      if (e.key === 'Backspace' && !b.value && i > 0) boxes[i - 1].focus();
    });
    b.addEventListener('paste', e => {
      const t = (e.clipboardData.getData('text') || '').replace(/\D/g, '').slice(0, 6);
      if (!t) return;
      e.preventDefault();
      [...t].forEach((c, j) => boxes[j].value = c);
      boxes[Math.min(t.length, 5)].focus();
    });
  });

  otpForm.addEventListener('submit', e => {
    e.preventDefault();
    const code = boxes.map(b => b.value).join('');
    if (code.length < 6) { $('otpErr').textContent = 'Enter all 6 digits.'; return; }
    if (code !== DEMO_CODE) { $('otpErr').textContent = "That code isn't right. Check it and try again."; return; }
    $('otpErr').textContent = '';
    clearInterval(countdown); show(done);
    // TODO: redirect to chats, e.g. location.href = '/chats';
  });

  function startTimer() {
    secs = 30; $('resend').classList.add('hidden'); tick();
    clearInterval(countdown);
    countdown = setInterval(() => { secs--; tick(); }, 1000);
  }
  function tick() {
    if (secs <= 0) {
      clearInterval(countdown); $('timer').textContent = ''; $('resend').classList.remove('hidden');
    } else $('timer').textContent = `Resend code in 0:${String(secs).padStart(2, '0')}`;
  }
  $('resend').addEventListener('click', () => { boxes.forEach(b => b.value = ''); boxes[0].focus(); startTimer(); });
  $('editNum').addEventListener('click', () => { clearInterval(countdown); show(phoneForm); $('phone').focus(); });
</script>
